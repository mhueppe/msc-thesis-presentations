\begin{frame}{Embeddings}
	\begin{columns}[T, totalwidth=\linewidth]
		% Left column (figure)
		\begin{column}{0.75\linewidth}
			\textbf{Sequence Embedding} \\
			\begin{itemize}
				\item ESM based from \cite{Rives2021}
				\item trained on 250m protein sequences 
				\item biochemical properties of amino acids to remote homology of proteins 
				\item deep Transformer learned using masked language modeling objective 
			\end{itemize}
			\textbf{Surface Embedding}
			\begin{itemize}
				\item trained on 200k protein structures
				\item surface sampled using dMaSif \parencite{Sverrisson2020}
