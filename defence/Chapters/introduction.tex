\subsection{Protein Crystallization}
\begin{frame}{Protein Crystallization}
    \begin{columns}[T]
        \column{0.59\textwidth}
        \begin{itemize}[<+->]
            \item The relation between \textbf{protein} and \textbf{precipitant concentration} defines different protein zones
            \item \textbf{Undersaturation:} protein stays dissolved
            \item \textbf{Nucleation zone:} new crystal nuclei can form {\tiny (cf. \ref{frame:NucleationBarrier})}
            \item \textbf{Metastable zone:} crystals can grow, but no spontaneous nucleation  {\tiny (cf. \ref{frame:NucleationBarrier})}
            \item \textbf{Precipitation zone:} excessive supersaturation leads to uncontrolled aggregation or many small crystals
            \item When a crystal is obtained, X-ray diffraction is used to determine the structure of the protein  {\tiny (cf. \ref{frame:Compounds})}
        \end{itemize}
