\begin{frame}
  \frametitle{Ein Titel\ldots\\			% optional title
	kann mehrzeilig sein}
  \framesubtitle{Optionaler Untertitel}		% optional subtitle
  Der Text ist immer im \alert{Frames} zusammengefasst;
  diese können (optional) Titel und Untertitel haben.

  Der eigentliche Text wird wie gewohnt mit \LaTeX{}-Befehlen gesetzt,
  als Fließtext oder meist in Form von
  \texttt{itemize}- oder \texttt{enumerate}"=Umgebungen.
\end{frame}

\begin{frame}
  \frametitle{\insertsubsection}
  Die Gliederung (\texttt{section}/\texttt{subsection}) wirkt sich nur
  auf das Inhaltsverzeichnis und die Kopfzeile aus, der
  \alert{\texttt{frametitle}} muss explizit gesetzt werden (wie oben).
  \bvskip

  Für Vorlesungen gibt es noch weitere Gliederungen (s. Doku.).
  \begin{itemize}
  \item \texttt{part} mit eigenen Inhaltsverzeichnissen und Titelseiten
	(\texttt{partpage})
  \item \texttt{lecture} mit include-Option\ldots
  \end{itemize}

  \begin{block}{Tipp}
  Nach meiner Erfahrung kommt man aber auch bei großen Vorlesungen,
  beispielsweise "`Rechnerstrukturen"' mit über 1\,400 Folien,
  ohne weitere Gliederungen aus. \texttt{section} und \texttt{subsection}
  mit Wiederholung des Inhaltsverzeichnis' durch Optionen wie
  \texttt{secToc}, etc. -- siehe Folie \ref{options} -- genügen.
  \end{block}
\end{frame}

\section{Overlays}			%-------------------------------------

\begin{frame}
  \frametitle{\insertsection}
  Die Verwendung von Overlays, aufklappenden Listen, etc.\ ist sicherlich
  Geschmackssache\ldots{} oft wird es extensiv benutzt, ich persönlich finde
  das scheußlich, hier aber trotzdem eine kleine Demo der Möglichkeiten.\bvskip

  Overlays können erzeugt werden mit
  \begin{itemize}
  \item dem \texttt{pause}-Befehl:
    \begin{itemize}
    \item eins
      \pause
    \item zwei
    \end{itemize}
  \item
    mit Overlay Spezifikationen:
    \begin{itemize}
    \item<3-> drei
    \item<4-> vier
    \end{itemize}
  \item
    oder mit \texttt{uncover}:
    \begin{itemize}
      \uncover<5->{\item fünf}
      \uncover<6->{\item sechs}
    \end{itemize}
  \end{itemize}
\end{frame}

\begin{frame}
  \frametitle{Effekte}
  Effekte durch Angabe der Frames bei den Befehlen: \texttt{<nr>}\bvskip

  \textbf{Dieser Text ist auf allen drei Folien fett.}
  \textbf<2>{Dieser Text ist nur auf der zweiten Folie fett.}
  \textbf<3>{Dieser Text ist nur auf der dritten Folie fett.}

  \only<1>{Nur auf der ersten Folie vorhanden.}
  \only<2>{Nur auf der zweiten Folie vorhanden.}

  \color<2-3>[rgb]{0,0,1} Blauer Text auf den Folien 2 und 3, sonst schwarz.
\end{frame}

\begin{frame}
  \frametitle{mehr Effekte}
  Auf Folie eins.
  \onslide<2-3>
  Auf Folie zwei und drei.
  \begin{itemize}
    \item Auch auf Folie zwei und drei.
    \onslide+<4->
    \item Ab Folie vier.
  \end{itemize}
  Ab Folie vier.
  \onslide
  Auf allen Folien.
  \bvskip

  \temporal<3-4>{Folie 1, 2}{Folie 3, 4}{Folie 5, 6, 7, ...}
  \temporal<3,5>{Folie 1, 2, 4}{Folie 3, 5}{Folie 6, 7, 8, ...}
  \bvskip

  \onslide<1>{Folie1: wie nächster Befehl.}
  \uncover<1>{Folie1: wie voriger Befehl.}

  \onslide+<2>{Folie2: wie nächster Befehl.}
  \visible<2>{Folie2: wie voriger Befehl.}

  \onslide*<3>{Folie3: wie nächster Befehl.}
  \only<3>{Folie3: wie voriger Befehl.}
\end{frame}

\begin{frame}
  Meist werden Overlays aber einfach benutzt:
  \begin{itemize}
   \item<1-> erster Punkt, auf allen Folien.\hfill{\small\texttt{<1->}}
   \item<2-> zweiter Punkt, ab Folie 2.\hfill{\small\texttt{<2->}}
   \item<2-> dritter Punkt, auch ab Folie 2.\hfill{\small\texttt{<2->}}
   \item<3-> vierter Punkt, ab Folie 3.\hfill{\small\texttt{<3->}}
  \end{itemize}
  \bvskip

  Hier etwas komplizierter mit \texttt{alert}:
  \begin{enumerate}
    \item<3-| alert@3>[0.] Der "`Null"'-Punkt, wird zuletzt gezeigt.
    \item<1-| alert@1> Der erste Punkt.
    \item<2-| alert@2> Der Zweite Punkt.
  \end{enumerate}
\end{frame}

\begin{frame}
  Automatische Overlays, mit Hervorhebung
  \begin{enumerate}[<+-| alert@+>]
    \item eins
    \item zwei
    \item drei
  \end{enumerate}
  \bvskip

  \ldots{} mit Hervorhebung einzelner Wörter:
  \begin{itemize}[<+->]
    \item Das ist \alert<.>{wichtig}.
    \item \alert<.>{Dieser} und \alert<.>{jener} Punkt aber auch.
    \item Und auch \alert<.>{hier} wird \texttt{alert} benutzt.
  \end{itemize}
  \bvskip

  \begin{block}{Blöcke}<2-3>
    Overlays funktionieren auch für Umgebungen \texttt{block},
    \texttt{example}, \texttt{theorem}, \texttt{proof}\ldots
  \end{block}
\end{frame}

\begin{frame}[label=hyperOne]
  \frametitle{\insertsubsection}
  Sprünge zwischen Frames und Overlays sind mit den Befehlen
  \texttt{hyperlink}, \texttt{hypertarget} / \texttt{label} und
  verschiedenen \texttt{button}-Befehlen möglich.

  \begin{description}
  \item[\texttt{beamerbutton}]\hfill
	\hyperlink{jumptofirst}{\beamerbutton{zu Overlay 1}}%
	\hypertarget<1>{jumptofirst}{}
  \item[\texttt{beamergotobutton}]\hfill
	\hyperlink{hyperOne<2>}{\beamergotobutton{zu Overlay 2}}
  \item[\texttt{beamerskipbutton}]\hfill
	\hyperlink{hyperTwo}{\beamerskipbutton{zur nächsten Folie}}
  \end{description}

  \begin{itemize}[<+->]
    \item Overlay 1
    \item Overlay 2
    \item Overlay 3
  \end{itemize}
\end{frame}

\begin{frame}[label=hyperTwo]
  \frametitle{\insertsubsection}
  \begin{description}
  \item[\texttt{beamerreturnbutton}]\hfill
	\hyperlink{hyperOne<3>}{\beamerreturnbutton{zur vorigen Folie,
	Overlay 3}}
  \end{description}

  Daneben gibt es noch verschiedene vordefinierte Hyperlinks, die für die
  Dokumentennavigation benutzt werden können,
  siehe \texttt{beamer}"=Dokumentation.
\end{frame}

\section{Multimedia}		%-------------------------------------

\begin{frame}[containsverbatim]
  \frametitle{Audio / Video}
  Die Integration von "`Multimedia-Inhalten"' ist leider immer noch abhängig
  vom Betriebssystem, der Einstellung des PDF-Viewers, externen Programmen etc.
  \bvskip

  Zu \texttt{beamer} gehört ein Paket \alert{\texttt{multimedia}},
  das die Befehle \texttt{movie} und \texttt{sound} bereitstellt
  (siehe Doku.). Während das früher nicht stabil lief, scheint
  das inzwischen (unter Linux: Ubuntu, SUSE) zu funktionieren.

  \begin{block}{Tipp}
  Ich hatte da bisher gute Erfahrungen mit dem Start als externes Programm,
  was auch im Vollbildmodus zuverlässig funktionierte:
  {\small\begin{verbatim}
  \movie[externalviewer]%
    {\includegraphics[height=35mm]{image/race2}}%
    {video/RACE/pr2-experience.mp4}
  \end{verbatim}}
  \end{block}
\end{frame}

\begin{frame}
  \frametitle{Externe Programme}
  Daneben ist es möglich beliebige externe Programme zu starten,
  dies funktioniert am einfachsten mit \alert{\texttt{href}} aus dem Paket
  \alert{\texttt{hyperref}}.\footnote{von \texttt{beamer} automatisch
	eingebunden}

  \begin{itemize}
  \item \href{run:/usr/bin/xeyes}{xeyes}
  \end{itemize}

  \begin{block}{Achtung}
    Wegen der benötigten Suchpfade, den Abhängigkeiten vom Programm etc.,
    sollte man Multimedia und externe Programme unbedingt vorher
    (in Ruhe) ausprobieren!
  \end{block}
\end{frame}

\begin{frame}[allowframebreaks]
  \label{options}%
  \frametitle{Optionen für \texttt{uhhBeamer}}
  \begin{itemize}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{german}}] definiert Strings für \texttt{titlepage}
	  und setzt die Sprache, bzw. die Silbentrennung, sonst \emph{english}
	\end{labeling}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{tams}}] TAMS Folien mit UHH-Farbschema und
	  vordefinierter Titelseite: \texttt{titlepage}
	\item[\texttt{\alert{cml}}] \txtbox{TAMS}{CML} Folien mit \ldots
	\end{labeling}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{cmlFund}}] "`funded by"' in CML Titelseite
	\end{labeling}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{uhhSans}}] Uni-Hausschrift "`The SansUHH"'
	\end{labeling}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{secHead}}] setzt Section-Namen als
	  Kopf"=Navigationszeile; funktioniert nur bei wenigen
	  \texttt{section}s: 7- 8

	  \alert{Voreinstellung für die Kopfzeile}\\
	  {\small\begin{tabular}{ll}
	  linksbündig	& \angOp{section}$[$ - \angOp{subsection}$]$\\
	  rechtsbündig	& \angOp{shorttitle}
	  \end{tabular}}
	\end{labeling}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{secToc}}] Inhaltsverzeichnis vor jeder Section
	\item[\texttt{\alert{secTocA}}] \dqbox{Inhaltsverzeichnis} vor jeder
	  Section, vollständig
	\item[\texttt{\alert{subsecToc}}] \dqbox{Inhaltsverzeichnis} vor jeder
	  Subsection
	\end{labeling}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{secNum}}] (Sub-)\,Sections durchnummeriert
	\end{labeling}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{noTocNum}}] keine Nummerierung im
	  Inhaltsverzeichnis
	\end{labeling}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{fullstep}}] alle Listen werden einzeln
	  "`aufgeklappt"'
	\end{labeling}
  \item \begin{labeling}{\texttt{noFootPages}}
	\item[\texttt{\alert{noFoot}}] schaltet die Fußzeile komplett ab.

	  \alert{Voreinstellung für die Fußzeile}\\
	  {\small\begin{tabular}{ll}
	  linksbündig	& \angOp{shortauthor} - \angOp{foottitle}\\
	  		& \angOp{shortauthor} - \angOp{shorttitle}
				bei \alert{\texttt{secHead}}\\
	  rechtsbündig	& \angOp{page} / \angOp{total pages}
