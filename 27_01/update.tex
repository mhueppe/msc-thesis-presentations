\begin{frame}{Positive--Unlabeled (PU) Learning}
	
	\begin{itemize}
		\item \textbf{Problem setting:}
		We observe \emph{positive} samples ($y=+1$) and \emph{unlabeled} data.
		Negative labels ($y=-1$) are never directly observed. \citet{Bekker2020}
		
		\item \textbf{Goal:}
		Learn a binary classifier $f(x)$ that separates positives from negatives
		despite missing negative labels.
		
		\vspace{0.5em}
		
		
		\item \textbf{PU risk (no negative labels required):}
	$$R_{\text{PU}}(f)
	=
	\underbrace{
		\pi_{+}\,\mathbb{E}_{+}[\ell(f(x),+1)]
	}_{\text{learn what "pos" looks like}}
	+
	\underbrace{
		\mathbb{E}_{U}[\ell(f(x),-1)]
	}_{\text{discourage predicting pos everywhere}}
	-
	\underbrace{
		\pi_{+}\,\mathbb{E}_{+}[\ell(f(x),-1)]
	}_{\text{remove hidden-pos penalty}}$$
		
		\item \textbf{Intuition:}
		Treat unlabeled data as negative \emph{on average}, then subtract the hidden positives.
	\end{itemize}
	
\end{frame}
